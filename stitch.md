Logic & Formula Inti (Core Algorithms) ### 6.1 Formula Perhitungan BMI & Kategori (Asia-Pasifik) ```typescript export function calculateBMI(weightKg: number, heightCm: number) { const heightM = heightCm / 100; const bmi = Number((weightKg / (heightM * heightM)).toFixed(1)); let category: "Kurus" | "Normal" | "Gemuk" | "Obesitas" = "Normal"; let badgeColor = "emerald"; if (bmi < 18.5) { category = "Kurus"; badgeColor = "blue"; } else if (bmi <= 24.9) { category = "Normal"; badgeColor = "emerald"; } else if (bmi <= 26.9) { category = "Gemuk"; badgeColor = "amber"; } else { category = "Obesitas"; badgeColor = "rose"; } const minIdealWeight = Number((18.5 * heightM * heightM).toFixed(1)); const maxIdealWeight = Number((24.9 * heightM * heightM).toFixed(1)); return { bmi, category, badgeColor, minIdealWeight, maxIdealWeight }; } ``` ### 6.2 Algoritma Filter Keamanan Sendi (Joint Protection Filter) ```typescript export function filterSafeExercises( exercises: Exercise[], userTools: string[], userConditions: JointCondition[] ): { safe: Exercise[]; excluded: Exercise[] } { const safe: Exercise[] = []; const excluded: Exercise[] = []; for (const ex of exercises) { // 1. Cek apakah ada kontraindikasi yang dimiliki user const hasConflict = ex.contraindicated.some(cond => userConditions.includes(cond)); // 2. Cek apakah alat tersedia const hasEquipment = ex.requiredTools.length === 0 || ex.requiredTools.every(tool => userTools.includes(tool)); if (!hasConflict && hasEquipment) { safe.push(ex); } else { excluded.push(ex); } } return { safe, excluded }; } ``` --- ## 7.
